## Shot chart — field in team colours, dots, team colour scale

```js
import * as Plot from "@observablehq/plot";
import { loadLeague } from "@sportsdataverse/sdvplot";
import { logos, surface, teamColor } from "@sportsdataverse/sdvplot/plot";
import { toSurfaceFrame } from "@sportsdataverse/sporty";

await loadLeague("nfl");
// ESPN plays: default input columns x/y, x a 0-100 yardline; the frame spans the whole field.
const shots = toSurfaceFrame(plays, { from: "espn-football-0-100" });
const court = surface("nfl", { team: "KC" });
Plot.plot({
  ...court.scales,
  width: 940,
  color: teamColor("nfl", { values: shots.map((s) => s.team) }),
  marks: [
    ...court.marks,
    Plot.dot(shots, { x: "surface_x", y: "surface_y", fill: "team", r: 5 }),
    logos(shots, { league: "nfl", x: "surface_x", y: "surface_y", team: "team", height: 0.08 }),
  ],
});
```

## Standings — bars with logos on the axis

```js
import * as Plot from "@observablehq/plot";
import { loadLeague } from "@sportsdataverse/sdvplot";
import { axisLogos, teamColor } from "@sportsdataverse/sdvplot/plot";

await loadLeague("nfl");
Plot.plot({
  height: 300,
  marks: [
    Plot.barY(standings, { x: "team", y: "wins", fill: "team" }),
    axisLogos("x", { league: "nfl", height: 0.12 }),
  ],
  color: teamColor("nfl", { values: standings.map((s) => s.team) }),
});
```

## Team tiers

```js
import * as Plot from "@observablehq/plot";
import { loadLeague } from "@sportsdataverse/sdvplot";
import { teamTiers } from "@sportsdataverse/sdvplot/plot";

await loadLeague("nfl");
Plot.plot(teamTiers(rows, { league: "nfl", caption: "data: nflverse" }));
```

`rows` carry `team` plus `tier_no` (and optionally `tier_rank`).
